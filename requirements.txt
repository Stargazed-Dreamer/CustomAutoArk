# 运行时依赖
# Python 版本：3.10（也兼容 3.11）
#
# 安装：pip install -r requirements.txt
# 如需 GPU 加速，安装完本文件后请额外覆盖安装 torch（见文末说明）

# ==================== 核心依赖 ====================
opencv-python>=4.8
numpy>=1.24,<2.0          # paddlepaddle / paddleocr 当前对 numpy 2.x 兼容性较差
matplotlib>=3.7
PySide6>=6.5

# ==================== OCR ====================
# PaddlePaddle / PaddleOCR 是 OCR 识别的核心
# CPU 模式可直接安装；GPU 模式见文末
paddlepaddle>=2.5
paddleocr>=2.7

# ==================== 配置文件 ====================
# 注意：PyPI 上传的包名是 PyYAML，import 名是 yaml
# 早期 requirements.txt 中误写为 `yaml`，已修正
PyYAML>=6.0

# ==================== GPU 加速（可选） ====================
# 安装 CUDA 与 cuDNN 后覆盖安装支持 CUDA 的 torch / torchvision
# torch 当前支持 CUDA 11.8 / 12.4 / 12.6，请根据你的 CUDA 版本选择 index-url
#   CUDA 11.8: https://download.pytorch.org/whl/cu118
#   CUDA 12.4: https://download.pytorch.org/whl/cu124
#   CUDA 12.6: https://download.pytorch.org/whl/cu126
#
# 示例（CUDA 12.4）：
#   pip install --force-reinstall torch==2.6.0 torchvision==0.21.1 "numpy<2.0.0" \
#       --index-url https://download.pytorch.org/whl/cu124
#
# CUDA 下载：
#   CUDA 11.8 https://developer.nvidia.com/cuda-11-8-0-download-archive
#   CUDA 12.4 https://developer.nvidia.com/cuda-12-4-0-download-archive
#   CUDA 12.6 https://developer.nvidia.com/cuda-12-6-0-download-archive
#
# ==================== 常见问题 ====================
# 如遇 OpenMP 冲突（OMP Error），通常是 intel-openmp 与其它库冲突：
#   pip uninstall intel-openmp
# 缺失此库再重新安装：
#   pip install intel-openmp
